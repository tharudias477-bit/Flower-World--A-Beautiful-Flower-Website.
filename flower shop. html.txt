    <script>
        // Shopping Cart Data Array
        let cart = [];

        // Single Page Navigation Logic (Switches among 7 main sections)
        function showPage(pageId) {
            // Hide all sections
            const pages = document.querySelectorAll('.page-section');
            pages.forEach(page => page.classList.remove('active-page'));

            // Deactivate all navigation header links
            const navLinks = document.querySelectorAll('nav a');
            navLinks.forEach(link => link.classList.remove('active'));

            // Display target page section
            const selectedPage = document.getElementById('page-' + pageId);
            if (selectedPage) {
                selectedPage.classList.add('active-page');
            }

            // Highlight target header link
            const activeNav = document.getElementById('nav-' + pageId);
            if (activeNav) {
                activeNav.classList.add('active');
            }

            // Smooth scroll to top of page
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        // Toggle Shopping Cart Slide Panel
        function toggleCart() {
            document.getElementById('cart-modal').classList.toggle('open');
        }

        // Add Product Item to Cart
        function addToCart(title, price) {
            cart.push({ title, price });
            updateCartUI();
            toggleCart();
        }

        // Update Cart Display & Total Calculation
        function updateCartUI() {
            const cartItemsContainer = document.getElementById('cart-items');
            const cartCount = document.getElementById('cart-count');
            const cartTotalPrice = document.getElementById('cart-total-price');

            cartCount.innerText = cart.length;

            if (cart.length === 0) {
                cartItemsContainer.innerHTML = '<p style="text-align: center; color: #888;">Your cart is empty.</p>';
                cartTotalPrice.innerText = 'Rs. 0.00';
                return;
            }

            let html = '';
            let total = 0;

            cart.forEach((item, index) => {
                total += item.price;
                html += `
                    <div class="cart-item">
                        <div>
                            <h4>${item.title}</h4>
                            <small>Rs. ${item.price.toLocaleString()}.00</small>
                        </div>
                        <i class="fa-solid fa-trash" style="color: red; cursor: pointer;" onclick="removeFromCart(${index})"></i>
                    </div>
                `;
            });

            cartItemsContainer.innerHTML = html;
            cartTotalPrice.innerText = `Rs. ${total.toLocaleString()}.00`;
        }

        // Remove Single Item from Cart
        function removeFromCart(index) {
            cart.splice(index, 1);
            updateCartUI();
        }

        // Checkout Action Trigger
        function checkout() {
            if (cart.length === 0) {
                alert('Your cart is empty. Please add items to proceed.');
                return;
            }
            alert('Thank you! Your order has been placed successfully.');
            cart = [];
            updateCartUI();
            toggleCart();
        }

        // Form Submit Handler
        function handleContactSubmit(event) {
            event.preventDefault();
            alert('Thank you for contacting Flower World. We will get back to you shortly!');
        }
    </script>
